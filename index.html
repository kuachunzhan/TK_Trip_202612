Apps Script 完整後端程式碼 (Code.gs) 
// ==========================================
// 東京六天五夜好友之旅 - Google Sheets 同步 API
// 自動維護：行程規劃 (Itinerary)、購物清單 (ShoppingList)、美食清單 (FoodList)
// ==========================================

function doGet(e) {
  var action = (e && e.parameter && e.parameter.action) ? e.parameter.action : 'pull';
  var ss = SpreadsheetApp.getActiveSpreadsheet();

  if (action === 'ping') {
    return jsonOutput({ status: 'ok', message: 'Connected to Google Spreadsheet!', sheetName: ss.getName() });
  }

  var data = {
    hotel: readHotelSheet(ss),
    days: readItinerarySheet(ss),
    shopping: readShoppingSheet(ss),
    foods: readFoodSheet(ss),
    pulledAt: new Date().toISOString()
  };
  return jsonOutput({ status: 'ok', data: data });
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var raw = e.postData ? e.postData.contents : '{}';
    var payload = JSON.parse(raw);
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    if (payload.action === 'ping') {
      return jsonOutput({ status: 'ok', message: 'POST connection verified', sheetName: ss.getName() });
    }

    if (payload.hotel) writeHotelSheet(ss, payload.hotel);
    if (payload.days) writeItinerarySheet(ss, payload.days);
    if (payload.shopping) writeShoppingSheet(ss, payload.shopping);
    if (payload.foods) writeFoodSheet(ss, payload.foods);

    return jsonOutput({
      status: 'ok',
      syncedAt: new Date().toISOString(),
      counts: {
        days: payload.days ? payload.days.length : 0,
        shopping: payload.shopping ? payload.shopping.length : 0,
        foods: payload.foods ? payload.foods.length : 0
      }
    });
  } catch (err) {
    return jsonOutput({ status: 'error', message: err.toString() });
  } finally {
    lock.releaseLock();
  }
}

function getOrCreateSheet(ss, name, headers) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    sheet.appendRow(headers);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#e0e7ff');
    sheet.setFrozenRows(1);
  }
  return sheet;
}

function writeHotelSheet(ss, hotel) {
  var sheet = getOrCreateSheet(ss, '住宿資訊', ['住宿名稱', '日文地址', '最近車站', '住宿備註', '最後更新時間']);
  sheet.clearContents();
  sheet.appendRow(['住宿名稱', '日文地址', '最近車站', '住宿備註', '最後更新時間']);
  sheet.appendRow([hotel.name || '', hotel.address || '', hotel.station || '', hotel.note || '', new Date().toLocaleString('zh-TW')]);
}

function readHotelSheet(ss) {
  var sheet = ss.getSheetByName('住宿資訊');
  if (!sheet || sheet.getLastRow() < 2) return null;
  var row = sheet.getRange(2, 1, 1, 4).getValues()[0];
  return { name: row[0], address: row[1], station: row[2], note: row[3] };
}

function writeItinerarySheet(ss, days) {
  var headers = ['DayID', '日期', '當日主題', 'EventID', '時間', '行程名稱', '標籤', '交通方式', '備註', '已完成'];
  var sheet = getOrCreateSheet(ss, '行程規劃', headers);
  sheet.clearContents();
  sheet.appendRow(headers);
  var rows = [];
  days.forEach(function(d) {
    (d.events || []).forEach(function(ev) {
      rows.push([d.id, d.dateLabel, d.dayTitle, ev.id, ev.time, ev.title, ev.tag, ev.transit || '', ev.note || '', ev.done ? 'TRUE' : 'FALSE']);
    });
  });
  if (rows.length > 0) {
    sheet.getRange(2, 1, rows.length, headers.length).setValues(rows);
  }
}

function readItinerarySheet(ss) {
  var sheet = ss.getSheetByName('行程規劃');
  if (!sheet || sheet.getLastRow() < 2) return null;
  var values = sheet.getRange(2, 1, sheet.getLastRow() - 1, 10).getValues();
  var dayMap = {};
  values.forEach(function(r) {
    var dayId = String(r[0]);
    if (!dayMap[dayId]) {
      dayMap[dayId] = { id: dayId, dateLabel: String(r[1]), dayTitle: String(r[2]), events: [] };
    }
    dayMap[dayId].events.push({
      id: String(r[3]),
      time: String(r[4]),
      title: String(r[5]),
      tag: String(r[6]),
      transit: String(r[7]),
      note: String(r[8]),
      done: String(r[9]).toUpperCase() === 'TRUE'
    });
  });
  return Object.keys(dayMap).map(function(k) { return dayMap[k]; });
}

function writeShoppingSheet(ss, shopping) {
  var headers = ['ItemID', '商品名稱', '分類', '預計購買地點', '日幣價格', '已購買'];
  var sheet = getOrCreateSheet(ss, '購物清單', headers);
  sheet.clearContents();
  sheet.appendRow(headers);
  var rows = shopping.map(function(s) {
    return [s.id, s.name, s.category, s.store || '', Number(s.price) || 0, s.bought ? 'TRUE' : 'FALSE'];
  });
  if (rows.length > 0) {
    sheet.getRange(2, 1, rows.length, headers.length).setValues(rows);
  }
}

function readShoppingSheet(ss) {
  var sheet = ss.getSheetByName('購物清單');
  if (!sheet || sheet.getLastRow() < 2) return null;
  var values = sheet.getRange(2, 1, sheet.getLastRow() - 1, 6).getValues();
  return values.map(function(r) {
    return {
      id: String(r[0]),
      name: String(r[1]),
      category: String(r[2]),
      store: String(r[3]),
      price: Number(r[4]) || 0,
      bought: String(r[5]).toUpperCase() === 'TRUE'
    };
  });
}

function writeFoodSheet(ss, foods) {
  var headers = ['FoodID', '餐廳名稱', '日文副標', '區域', '地址', '招牌必點', '預計價位', '預約建議', '建議時段', '交通方式', '已造訪'];
  var sheet = getOrCreateSheet(ss, '美食清單', headers);
  sheet.clearContents();
  sheet.appendRow(headers);
  var rows = foods.map(function(f) {
    return [f.id, f.name, f.jpName || '', f.area, f.address || '', f.signature, f.priceRange || '', f.reservation || '', f.suggestedTime || '', f.transit || '', f.visited ? 'TRUE' : 'FALSE'];
  });
  if (rows.length > 0) {
    sheet.getRange(2, 1, rows.length, headers.length).setValues(rows);
  }
}

function readFoodSheet(ss) {
  var sheet = ss.getSheetByName('美食清單');
  if (!sheet || sheet.getLastRow() < 2) return null;
  var values = sheet.getRange(2, 1, sheet.getLastRow() - 1, 11).getValues();
  return values.map(function(r) {
    return {
      id: String(r[0]),
      name: String(r[1]),
      jpName: String(r[2]),
      area: String(r[3]),
      address: String(r[4]),
      signature: String(r[5]),
      priceRange: String(r[6]),
      reservation: String(r[7]),
      suggestedTime: String(r[8]),
      transit: String(r[9]),
      visited: String(r[10]).toUpperCase() === 'TRUE'
    };
  });
}

function jsonOutput(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
